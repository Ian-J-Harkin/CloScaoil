2. Unified Arbitrator ArchitectureThe new AmbiguityArbitrator will use a single completion method. This allows you to switch between openai/gpt-4o, anthropic/claude-3.5-sonnet, or any model on OpenRouter just by changing a configuration string.Pythonimport litellm
import os

class AmbiguityArbitrator:
    def __init__(self, api_key=None, provider="gemini", model_name="gemini-1.5-flash"):
        # LiteLLM reads keys from environment variables automatically
        # e.g., OPENAI_API_KEY, ANTHROPIC_API_KEY, OPENROUTER_API_KEY
        self.model = f"{provider}/{model_name}" if provider != "openai" else model_name
        
    def resolve(self, word, options, context_sentence):
        prompt = f"In this 1940s Irish sentence, is [?] '{word}' or one of {options}? Context: {context_sentence}"
        
        try:
            # LiteLLM standardizes the call for all providers
            response = litellm.completion(
                model=self.model,
                messages=[{"role": "user", "content": prompt}]
            )
            return response.choices[0].message.content.strip()
        except Exception as e:
            print(f"LLM Error: {e}")
            return None
